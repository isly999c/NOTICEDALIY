// ✅ v2.6 語音強化版：邊講邊出字、支援連續換氣、自動續聽不漏字 const micBtn = document.getElementById('micBtn'); const speechStatus = document.getElementById('speechStatus'); const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (SpeechRecognition) { const recognition = new SpeechRecognition(); recognition.lang = 'zh-TW'; recognition.continuous = true; // 允許停頓換氣，不會講一句就斷掉 recognition.interimResults = true; // 邊講邊即時顯示文字，防止整句消失 let isListening = false; let baseText = ''; let latestInterim = ''; recognition.onstart = () => { isListening = true; const input = document.getElementById('logInput'); baseText = input.value ? input.value + '\n' : ''; latestInterim = ''; micBtn.classList.add('recording'); micBtn.innerText = '🛑 正在收音中... (講完點擊結束)'; speechStatus.classList.remove('hidden'); speechStatus.innerText = '🎙️ 請開始說話，文字會即時浮現...'; }; recognition.onresult = (e) => { let finalTranscript = ''; let interimTranscript = ''; for (let i = 0; i < e.results.length; i++) { const transcript = e.results[i][0].transcript; if (e.results[i].isFinal) { finalTranscript += transcript + ' '; } else { interimTranscript += transcript; } } latestInterim = interimTranscript; const input = document.getElementById('logInput'); // 即時將「已確認文字 + 正在講的文字」寫入輸入框，絕對不會沒字！ input.value = baseText + finalTranscript + interimTranscript; input.scrollTop = input.scrollHeight; input.dispatchEvent(new Event('input')); if (interimTranscript) { speechStatus.innerText = `👂 聽取中：${interimTranscript}`; } else if (finalTranscript) { speechStatus.innerText = `✅ 已寫入文字，請繼續說或點擊結束...`; } }; recognition.onerror = (e) => { if (e.error === 'no-speech') { speechStatus.innerText = '⚠️ 沒有偵測到聲音，請靠近麥克風再說一次...'; } else if (e.error === 'not-allowed') { isListening = false; speechStatus.innerText = '❌ 麥克風權限被拒絕！請在瀏覽器網址列左側開啟麥克風權限，或直接使用手機鍵盤上的「🎤」語音輸入。'; } else { speechStatus.innerText = `⚠️ 語音提示 (${e.error})，若無法收音請直接點下方輸入框用手機鍵盤「🎤」輸入。`; } }; recognition.onend = () => { micBtn.classList.remove('recording'); micBtn.innerText = '🎤 按一下開始語音輸入 (邊講邊出字)'; if (isListening) { isListening = false; setTimeout(() => speechStatus.classList.add('hidden'), 3000); } }; micBtn.addEventListener('click', () => { if (isListening) { isListening = false; recognition.stop(); } else { try { recognition.start(); } catch (err) { recognition.stop(); } } }); } else { micBtn.innerText = '⌨️ 點擊下方輸入框，使用手機鍵盤「🎤」語音輸入'; micBtn.onclick = () => document.getElementById('logInput').focus(); }
